cmake_minimum_required(VERSION 3.22)
project(android_effect_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_executable(android_effect_tests test_procesador.cpp)
target_include_directories(android_effect_tests PRIVATE ../../app/src/main/cpp ${OpenCV_INCLUDE_DIRS})
target_link_libraries(android_effect_tests PRIVATE ${OpenCV_LIBS})
enable_testing()
add_test(NAME android_effect_tests COMMAND android_effect_tests)
